; FUNC1: caller cleans the stack (no `ret N` prefix)
FUNC1 PROC
    push ebp
    mov  ebp, esp
    mov  eax, [ebp+8]   ; first argument (4 bytes)
    add  eax, [ebp+12]  ; second argument (4 bytes)
    pop  ebp
    ret                 ; bare ret  -> caller pops arguments (cdecl)
FUNC1 ENDP

; FUNC2: callee cleans the stack (`ret 8` pops the 8 argument bytes)
FUNC2 PROC
    push ebp
    mov  ebp, esp
    mov  eax, [ebp+8]
    sub  eax, [ebp+12]
    pop  ebp
    ret  8              ; ret with operand -> callee cleans (stdcall)
FUNC2 ENDP
